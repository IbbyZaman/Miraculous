// List of scraper API templates to attempt in order function getScraperEndpoints(id, type, season, episode) { if (type === 'movie') { return [ `https://vidsrc.stream/m3u8/movie/${id}`, `https://api.vidsrc.icu/raw/movie/${id}`, `https://vidsrc.cc/v2/embed/movie/${id}` ]; } else { return [ `https://vidsrc.stream/m3u8/tv/${id}/${season}/${episode}`, `https://api.vidsrc.icu/raw/tv/${id}/${season}/${episode}`, `https://vidsrc.cc/v2/embed/tv/${id}/${season}/${episode}` ]; } } let currentScraperIndex = 0; async function loadMediaStream() { if (!activeMedia) return; const id = activeMedia.id; const season = seasonSelect.value || 1; const episode = episodeSelect.value || 1; if (hlsInstance) { hlsInstance.destroy(); hlsInstance = null; } if (activeServer === 'proxy_hls') { iframeFallbackPlayer.style.display = 'none'; iframeFallbackPlayer.src = ''; nativeVideoPlayer.style.display = 'block'; const scraperList = getScraperEndpoints(id, currentCategory, season, episode); // If we run out of direct scrapers to try, switch to embed backup if (currentScraperIndex >= scraperList.length) { console.warn("All direct HLS scrapers failed. Switching to embed iframe fallback..."); currentScraperIndex = 0; // Reset index const fallbackBtn = document.querySelectorAll('.server-btn')[1]; switchServer('vidsrc_sbs', fallbackBtn); return; } const currentTarget = scraperList[currentScraperIndex]; const proxiedUrl = `${PROXY_WORKER}?url=${encodeURIComponent(currentTarget)}`; if (Hls.isSupported()) { hlsInstance = new Hls({ manifestLoadingTimeOut: 5000, // Timeout after 5s if scraper hangs levelLoadingTimeOut: 5000 }); hlsInstance.loadSource(proxiedUrl); hlsInstance.attachMedia(nativeVideoPlayer); hlsInstance.on(Hls.Events.MANIFEST_PARSED, () => { nativeVideoPlayer.play().catch(() => {}); }); // SCRAPER FAILURE HANDLER (Tries next scraper in list) hlsInstance.on(Hls.Events.ERROR, (event, data) => { if (data.fatal) { console.warn(`Scraper #${currentScraperIndex + 1} failed. Trying next scraper...`); hlsInstance.destroy(); hlsInstance = null; currentScraperIndex++; // Increment to next scraper provider loadMediaStream(); // Retry stream loading } }); } else if (nativeVideoPlayer.canPlayType('application/vnd.apple.mpegurl')) { nativeVideoPlayer.src = proxiedUrl; nativeVideoPlayer.onerror = () => { currentScraperIndex++; loadMediaStream(); }; } } else { // Embed Server Mode currentScraperIndex = 0; // Reset index on manual server switch nativeVideoPlayer.pause(); nativeVideoPlayer.style.display = 'none'; iframeFallbackPlayer.style.display = 'block'; let url = ''; if (currentCategory === 'movie') { if (activeServer === 'vidsrc_sbs') url = `https://vidsrc.sbs/embed/movie/${id}`; else if (activeServer === 'vidsrc_pro') url = `https://vidsrc.pro/embed/movie/${id}`; else if (activeServer === 'autoembed') url = `https://player.autoembed.cc/embed/movie/${id}`; } else { if (activeServer === 'vidsrc_sbs') url = `https://vidsrc.sbs/embed/tv/${id}/${season}/${episode}`; else if (activeServer === 'vidsrc_pro') url = `https://vidsrc.pro/embed/tv/${id}/${season}/${episode}`; else if (activeServer === 'autoembed') url = `https://player.autoembed.cc/embed/tv/${id}/${season}/${episode}`; } iframeFallbackPlayer.src = url; } if (currentCategory === 'movie') { saveToHistory(activeMedia, 'movie'); } else { saveToHistory(activeMedia, 'tv', season, episode); } }
